📦 Simulation Code Snippets

test_sltu.c
int main() {
    unsigned int a = 5, b = 10, res;
    asm volatile ("sltu %0, %1, %2" : "=r"(res) : "r"(a), "r"(b));
    return res;  // res should be 1
}
test_srai.c
int main() {
    int val = -32, result;
    asm volatile ("srai %0, %1, 2" : "=r"(result) : "r"(val));
    return result;  // Arithmetic right shift
}
test_sw.c
int main() {
    int value = 42;
    int *addr = (int*)0x10000000;
    asm volatile("sw %0, 0(%1)" : : "r"(value), "r"(addr));
    return 0;
}
